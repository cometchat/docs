{/*
  The v1 → v2 upgrade notice, shared by every widget page.

  Only the prose lives here; each page supplies its own <Warning> or <Note> wrapper. Integration
  pages use <Warning>, overview pages <Note> — on an overview the reader is usually new and the
  notice does not apply to them.

  JSX, not Markdown: inside an exported component `**bold**`, backticks and `[text](/url)` render
  literally, so this uses <strong>, <code> and <a>. The {' '} before a tag at a line break is
  load-bearing — JSX drops whitespace adjacent to tags.
*/}

export const V1UpgradeBody = () => (
  <>
    <strong>Using the v1 widget (Chat Embed <code>@1.x.x</code>)?</strong> Widgets created in the

    upgrade, read{' '}

    JavaScript may need updating.

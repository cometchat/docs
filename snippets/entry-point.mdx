{/*
  The docs entry point: one line under the H1, one link to the Start-here guide for the
  stack of the page it sits on. The wording lives here and nowhere else, so it changes in
  one place (task T-docs-entry.05).

  Props
    stack     the stack's name, for example "React". Leave it out on a page with no single
              stack; the line then points at the page where a reader picks one.
    href      the Start-here guide for that stack.
    campaign  the docs section the page sits in. It becomes utm_campaign.

  Every link carries utm_source=docs and utm_medium=entry, so a signup that starts on a
  docs page can be counted (task T-docs-entry.07).

  JSX, not Markdown: inside an exported component, Markdown renders literally. The {' '}
  keeps the space that JSX drops next to a tag.
*/}

export const EntryPoint = ({ stack, href = "/chat-call", campaign = "docs" }) => (
  <p>
    New to CometChat?{' '}
    <a href={`${href}?utm_source=docs&utm_medium=entry&utm_campaign=${campaign}`}>
      {stack ? `Start with the ${stack} integration guide` : "Pick your stack and start building"}
    </a>
    .
  </p>
);

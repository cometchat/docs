Build a custom participant list UI that displays real-time participant information with full control over layout and interactions. This guide demonstrates how to hide the default participant list and create your own using participant events and actions.

## Overview

The SDK provides participant data through events, allowing you to build custom UIs for:
- Participant roster with search and filtering
- Custom participant cards with role badges or metadata
- Moderation dashboards with quick access to controls
- Attendance tracking and engagement monitoring

## Prerequisites

- CometChat Calls SDK installed and initialized
- Active call session (see [Join Session](/calls/flutter/join-session))
- Familiarity with [Actions](/calls/flutter/actions) and [Events](/calls/flutter/events)

## Step 1: Hide Default Participant List

Configure session settings to hide the default participant list button:

```dart
final sessionSettings = (SessionSettingsBuilder()
    ..hideParticipantListButton(true))
    .build();
```

## Step 2: Create Participant List Widget

Build a Flutter widget for displaying participants. Instead of Android's RecyclerView and XML layouts, Flutter uses `ListView` and widget composition:

```dart
class ParticipantListPanel extends StatelessWidget {
  final List<Participant> participants;
  final String searchQuery;
  final ValueChanged<String> onSearchChanged;
  final ValueChanged<Participant> onMute;
  final ValueChanged<Participant> onPauseVideo;
  final ValueChanged<Participant> onPin;
  final VoidCallback onClose;

  const ParticipantListPanel({
    super.key,
    required this.participants,
    required this.searchQuery,
    required this.onSearchChanged,
    required this.onMute,
    required this.onPauseVideo,
    required this.onPin,
    required this.onClose,
  });

  List<Participant> get filteredParticipants {
    if (searchQuery.isEmpty) return participants;
    return participants
        .where((p) => p.name.toLowerCase().contains(searchQuery.toLowerCase()))
        .toList();
  }

  @override
  Widget build(BuildContext context) {
    final filtered = filteredParticipants;

    return Container(
      width: 300,
      color: const Color(0xFFF5F5F5),
      padding: const EdgeInsets.all(16),
      child: Column(
        children: [
          // Header
          Row(
            children: [
              Expanded(
                child: Text(
                  "Participants (${filtered.length})",
                  style: const TextStyle(
                    fontSize: 18,
                    fontWeight: FontWeight.bold,
                  ),
                ),
              ),
              IconButton(
                icon: const Icon(Icons.close),
                onPressed: onClose,
              ),
            ],
          ),
          const SizedBox(height: 8),
          // Search Bar
          TextField(
            onChanged: onSearchChanged,
            decoration: const InputDecoration(
              hintText: "Search participants...",
              filled: true,
              fillColor: Colors.white,
              border: OutlineInputBorder(),
            ),
          ),
          const SizedBox(height: 16),
          // Participant List
          Expanded(
            child: ListView.builder(
              itemCount: filtered.length,
              itemBuilder: (context, index) {
                final view = filtered[index];
                return _ParticipantTile(
                  view: view,
                  onMute: () => onMute(view),
                  onPauseVideo: () => onPauseVideo(view),
                  onPin: () => onPin(view),
                );
              },
            ),
          ),
        ],
      ),
    );
  }
}

/// `Participant` carries identity and usage data only — it has no mute, video,
/// hand-raise, screen-share or pinned flags. Hold that state alongside it and
/// update it from `ParticipantEventListeners` events (Step 3).
class ParticipantView {
  ParticipantView(this.participant);

  Participant participant;
  bool muted = false;
  bool videoPaused = false;
  bool presenting = false;
  bool handRaised = false;
  bool pinned = false;

  String get mid => participant.mid ?? "";
  String get displayName => participant.name ?? "Unknown";
}

class _ParticipantTile extends StatelessWidget {
  final ParticipantView view;
  final VoidCallback onMute;
  final VoidCallback onPauseVideo;
  final VoidCallback onPin;

  const _ParticipantTile({
    required this.view,
    required this.onMute,
    required this.onPauseVideo,
    required this.onPin,
  });

  String get statusText {
    final parts = <String>[];
    if (view.muted) parts.add("🔇 Muted");
    if (view.videoPaused) parts.add("📹 Video Off");
    if (view.presenting) parts.add("🖥️ Presenting");
    if (view.handRaised) parts.add("✋ Hand Raised");
    if (view.pinned) parts.add("📌 Pinned");
    return parts.isEmpty ? "Active" : parts.join(" • ");
  }

  @override
  Widget build(BuildContext context) {
    final avatar = view.participant.avatar;
    final name = view.displayName;

    return ListTile(
      leading: CircleAvatar(
        backgroundImage: avatar != null ? NetworkImage(avatar) : null,
        child: avatar == null ? Text(name[0].toUpperCase()) : null,
      ),
      title: Text(
        name,
        style: const TextStyle(fontWeight: FontWeight.bold),
      ),
      subtitle: Text(statusText, style: const TextStyle(fontSize: 12)),
      trailing: Row(
        mainAxisSize: MainAxisSize.min,
        children: [
          IconButton(
            icon: const Icon(Icons.mic_off, size: 20),
            onPressed: onMute,
            color: view.muted ? Colors.grey : null,
          ),
          IconButton(
            icon: const Icon(Icons.videocam_off, size: 20),
            onPressed: onPauseVideo,
            color: view.videoPaused ? Colors.grey : null,
          ),
          IconButton(
            icon: const Icon(Icons.push_pin, size: 20),
            onPressed: onPin,
            color: view.pinned ? Colors.blue : Colors.grey,
          ),
        ],
      ),
    );
  }
}
```

## Step 3: Implement Participant Events

Listen for participant updates and handle actions in your call screen:

```dart
class _ParticipantEventListener extends ParticipantEventListeners {
  _ParticipantEventListener(this.onChanged);

  /// Called after any change so the widget can `setState`.
  final void Function(List<ParticipantView>) onChanged;

  /// Keyed on `mid`, which is what identifies a participant within a session.
  final Map<String, ParticipantView> _views = {};

  /// `mid` of the pinned participant, or null. The SDK reports no pinned state,
  /// so we own it.
  String? pinnedMid;

  List<ParticipantView> get views => _views.values.toList();

  ParticipantView _viewFor(Participant participant) {
    final mid = participant.mid ?? "";
    final view = _views.putIfAbsent(mid, () => ParticipantView(participant));
    view.participant = participant;
    return view;
  }

  @override
  void onParticipantListChanged(List<Participant> updatedParticipants) {
    final seen = <String>{};
    for (final participant in updatedParticipants) {
      seen.add(participant.mid ?? "");
      _viewFor(participant);
    }
    _views.removeWhere((mid, _) => !seen.contains(mid));
    onChanged(views);
  }

  @override
  void onParticipantJoined(Participant participant) {
    _viewFor(participant);
    onChanged(views);
  }

  @override
  void onParticipantLeft(Participant participant) {
    final mid = participant.mid ?? "";
    _views.remove(mid);
    if (pinnedMid == mid) pinnedMid = null;
    onChanged(views);
  }

  @override
  void onParticipantAudioMuted(Participant participant) {
    _viewFor(participant).muted = true;
    onChanged(views);
  }

  @override
  void onParticipantAudioUnmuted(Participant participant) {
    _viewFor(participant).muted = false;
    onChanged(views);
  }

  @override
  void onParticipantVideoPaused(Participant participant) {
    _viewFor(participant).videoPaused = true;
    onChanged(views);
  }

  @override
  void onParticipantVideoResumed(Participant participant) {
    _viewFor(participant).videoPaused = false;
    onChanged(views);
  }

  @override
  void onParticipantHandRaised(Participant participant) {
    _viewFor(participant).handRaised = true;
    onChanged(views);
  }

  @override
  void onParticipantHandLowered(Participant participant) {
    _viewFor(participant).handRaised = false;
    onChanged(views);
  }

  @override
  void onParticipantStartedScreenShare(Participant participant) {
    _viewFor(participant).presenting = true;
    onChanged(views);
  }

  @override
  void onParticipantStoppedScreenShare(Participant participant) {
    _viewFor(participant).presenting = false;
    onChanged(views);
  }

  void setPinned(String? mid) {
    for (final view in _views.values) {
      view.pinned = view.mid == mid;
    }
    pinnedMid = mid;
    onChanged(views);
  }
}

class _CallScreenState extends State<CallScreen> {
  List<ParticipantView> participants = [];
  String searchQuery = "";
  bool isParticipantPanelVisible = false;

  late final _ParticipantEventListener _participantListener;

  @override
  void initState() {
    super.initState();
    _setupParticipantListener();
  }

  void _setupParticipantListener() {
    _participantListener = _ParticipantEventListener((views) {
      if (mounted) setState(() => participants = views);
    });
    CallSession.getInstance()?.addParticipantEventListener(_participantListener);
  }

  void _onMuteParticipant(ParticipantView view) async {
    final uid = view.participant.uid;
    if (uid == null) return;
    await CallSession.getInstance()?.muteParticipant(uid);
  }

  void _onPauseParticipantVideo(ParticipantView view) async {
    final uid = view.participant.uid;
    if (uid == null) return;
    await CallSession.getInstance()?.pauseParticipantVideo(uid);
  }

  void _onPinParticipant(ParticipantView view) async {
    final session = CallSession.getInstance();
    if (view.pinned) {
      await session?.unPinParticipant();
      _participantListener.setPinned(null);
      return;
    }
    final uid = view.participant.uid;
    if (uid == null) return;
    await session?.pinParticipant(uid);
    _participantListener.setPinned(view.mid);
  }

  @override
  void dispose() {
    CallSession.getInstance()?.removeParticipantEventListener(_participantListener);
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Stack(
        children: [
          // Call view
          if (callWidget != null) callWidget!,

          // Participant panel (slides in from right)
          if (isParticipantPanelVisible)
            Positioned(
              right: 0,
              top: 0,
              bottom: 0,
              child: ParticipantListPanel(
                participants: participants,
                searchQuery: searchQuery,
                onSearchChanged: (query) {
                  setState(() => searchQuery = query);
                },
                onMute: _onMuteParticipant,
                onPauseVideo: _onPauseParticipantVideo,
                onPin: _onPinParticipant,
                onClose: () {
                  setState(() => isParticipantPanelVisible = false);
                },
              ),
            ),
        ],
      ),
      floatingActionButton: FloatingActionButton(
        onPressed: () {
          setState(() => isParticipantPanelVisible = !isParticipantPanelVisible);
        },
        child: const Icon(Icons.people),
      ),
    );
  }
}
```

<Note>
Flutter listeners are not lifecycle-aware. You must manually remove listeners in `dispose()` to prevent memory leaks.
</Note>

## Complete Example

Here's the full implementation with all components:

```dart
import 'package:cometchat_calls_sdk/cometchat_calls_sdk.dart';
import 'package:flutter/material.dart';

// ParticipantView and _ParticipantEventListener are the versions defined in
// Steps 2 and 3 above.

class CallScreen extends StatefulWidget {
  final String sessionId;

  const CallScreen({super.key, required this.sessionId});

  @override
  State<CallScreen> createState() => _CallScreenState();
}

class _CallScreenState extends State<CallScreen> {
  Widget? callWidget;
  List<ParticipantView> participants = [];
  String searchQuery = "";
  bool isParticipantPanelVisible = false;

  late final _ParticipantEventListener _participantListener;

  @override
  void initState() {
    super.initState();
    _setupParticipantListener();
    _joinCall();
  }

  void _joinCall() {
    final sessionSettings = (SessionSettingsBuilder()
        ..hideParticipantListButton(true)
        ..setTitle("Team Meeting"))
        .build();

    CometChatCalls.joinSession(
      sessionId: widget.sessionId,
      sessionSettings: sessionSettings,
      onSuccess: (Widget? widget) {
        setState(() => callWidget = widget);
      },
      onError: (CometChatCallsException e) {
        debugPrint("Failed to join: ${e.message}");
      },
    );
  }

  void _setupParticipantListener() {
    _participantListener = _ParticipantEventListener((views) {
      if (mounted) setState(() => participants = views);
    });
    CallSession.getInstance()?.addParticipantEventListener(_participantListener);
  }

  void _onMuteParticipant(ParticipantView view) async {
    final uid = view.participant.uid;
    if (uid == null) return;
    await CallSession.getInstance()?.muteParticipant(uid);
  }

  void _onPauseParticipantVideo(ParticipantView view) async {
    final uid = view.participant.uid;
    if (uid == null) return;
    await CallSession.getInstance()?.pauseParticipantVideo(uid);
  }

  void _onPinParticipant(ParticipantView view) async {
    final session = CallSession.getInstance();
    if (view.pinned) {
      await session?.unPinParticipant();
      _participantListener.setPinned(null);
      return;
    }
    final uid = view.participant.uid;
    if (uid == null) return;
    await session?.pinParticipant(uid);
    _participantListener.setPinned(view.mid);
  }

  @override
  void dispose() {
    CallSession.getInstance()?.removeParticipantEventListener(_participantListener);
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Stack(
        children: [
          if (callWidget != null) callWidget!,
          if (isParticipantPanelVisible)
            Positioned(
              right: 0,
              top: 0,
              bottom: 0,
              child: ParticipantListPanel(
                participants: participants,
                searchQuery: searchQuery,
                onSearchChanged: (query) {
                  setState(() => searchQuery = query);
                },
                onMute: _onMuteParticipant,
                onPauseVideo: _onPauseParticipantVideo,
                onPin: _onPinParticipant,
                onClose: () {
                  setState(() => isParticipantPanelVisible = false);
                },
              ),
            ),
        ],
      ),
      floatingActionButton: FloatingActionButton(
        onPressed: () {
          setState(() => isParticipantPanelVisible = !isParticipantPanelVisible);
        },
        child: const Icon(Icons.people),
      ),
    );
  }
}
```

<Accordion title="Participant Object Reference">
| Property | Type | Description |
|----------|------|-------------|
| `uid` | `String?` | CometChat user ID |
| `name` | `String?` | Display name |
| `avatar` | `String?` | URL of avatar image |
| `mid` | `String?` | Media ID — identifies this participant within the session |
| `deviceId` | `String?` | Device the participant joined from |
| `state` | `String?` | Participant's current state in the session |
| `isJoined` | `bool?` | Whether the participant is currently joined |
| `joinedAt` | `int?` | Timestamp when the participant joined |
| `leftAt` | `int?` | Timestamp when the participant left |
| `totalAudioMinutes` | `double?` | Audio minutes consumed by this participant |
| `totalVideoMinutes` | `double?` | Video minutes consumed by this participant |
| `totalDurationInMinutes` | `double?` | Total minutes this participant spent in the session |

Every field is nullable. There are no mute, video, hand-raise, screen-share or pinned flags —
that is why this page keeps a `ParticipantView` alongside each `Participant`.
</Accordion>
